# Cazador de Desperdicio — versiones fijas.
# Python 3.12. Instalar dentro de un venv:  pip install -r requirements.txt
#
# `strands-agents[anthropic]` arrastra el SDK `anthropic`, que es el que usa el
# camino de Claude API directa. El camino de Bedrock usa boto3.
strands-agents[anthropic]==1.50.2
anthropic==0.120.2
boto3==1.43.66

# Cálculo. scipy entra como dependencia de statsmodels, pero se fija explícito
# porque `calculos.py` la usa directo para los t-tests de los lotes.
pandas==3.0.5
numpy==2.5.1
statsmodels==0.14.6
scipy==1.18.0

# Interfaz. altair ya viene con streamlit y es la que dibuja los seis gráficos:
# se fija explícita para que el taller sea reproducible, no porque sea un extra.
streamlit==1.61.1
altair==6.2.2

# El reporte descargable lleva las gráficas incrustadas como SVG. `vl-convert`
# traduce el Vega-Lite de Altair a SVG **sin abrir un navegador** (es una
# biblioteca en Rust), que es lo que permite generar el archivo dentro del proceso
# de Streamlit. Sin esto, un reporte con gráficas exigiría un navegador headless.
vl-convert-python==1.9.0

python-dotenv==1.2.2
# El reporte descargable es un PDF. `fpdf2` es Python puro —sin librerias
# nativas que compilar, a diferencia de weasyprint, que pide pango y cairo— y
# maqueta A4 con margenes, saltos de pagina y numeracion. Las graficas entran
# como PNG generado por `vl-convert` (que ya estaba), no como SVG: fpdf dibuja
# SVG como vectores pero no sabe escribir su texto, y un grafico sin etiquetas
# de eje no sirve. Sigue sin hacer falta un navegador headless.
fpdf2==2.8.5


# --------------------------------------------------------------------------- #
# Dependencias transitivas fijadas a mano. No son extras: sin ellas el arranque
# del README no funciona en una parte de las máquinas del taller.
#
# `cryptography` llega por `strands-agents` -> `mcp` -> `pyjwt[crypto]`. Sin
# fijarla, pip resuelve la 50.x, que para macOS x86_64 (Intel) publica SOLO el
# sdist: intenta compilarla con Rust y OpenSSL y el `pip install -r
# requirements.txt` de una máquina limpia se cae con "Failed building wheel for
# cryptography". 48.0.1 es la última con rueda `macosx_10_9_universal2`, válida
# en Intel y en Apple Silicon; en Windows y Linux no cambia nada.
cryptography==48.0.1
